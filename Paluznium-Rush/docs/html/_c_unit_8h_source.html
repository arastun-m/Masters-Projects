79 void calculateOutputs(double paluszniumFeed, double gormaniumFeed, double wasteFeed) {
80 // Ensure minimum flow rate to prevent division by zero
81 paluszniumFeed = std::max(paluszniumFeed, MIN_FLOW_RATE);
82 gormaniumFeed = std::max(gormaniumFeed, MIN_FLOW_RATE);
83 wasteFeed = std::max(wasteFeed, MIN_FLOW_RATE);
84
85 // Calculate volumetric feed rate (sum of Fi/ρ)
86 double sumFiOverRho = (paluszniumFeed + gormaniumFeed + wasteFeed) / DENSITY;
87
88 // Calculate residence time (τ)
89 double tau = (SOLIDS_FRACTION * m_volume) / sumFiOverRho;
90
91 // Calculate recoveries to concentrate
92 double R_palusznium = (k_palusznium * tau) / (1.0 + k_palusznium * tau);
93 double R_gormanium = (k_gormanium * tau) / (1.0 + k_gormanium * tau);
94 double R_waste = (k_waste * tau) / (1.0 + k_waste * tau);
95
96 // Calculate output flowrates to concentrate
97 m_paluszniumConcentrate = paluszniumFeed * R_palusznium;
98 m_gormaniumConcentrate = gormaniumFeed * R_gormanium;
99 m_wasteConcentrate = wasteFeed * R_waste;
100
101 // Calculate tailings as remainder
102 m_paluszniumTailings = paluszniumFeed - m_paluszniumConcentrate;
103 m_gormaniumTailings = gormaniumFeed - m_gormaniumConcentrate;
104 m_wasteTailings = wasteFeed - m_wasteConcentrate;
